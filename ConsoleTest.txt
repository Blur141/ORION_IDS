const ws = new WebSocket("ws://localhost:8000/ws");

ws.onopen = () => console.log("✅ Connected to IDS!");

ws.onmessage = (event) => {
  const data = JSON.parse(event.data);
  
  if (data.type === "packet") {
    console.log("📦 Packet:", data.data.src_ip, "→", data.data.dst_ip, 
                "[" + data.data.category + "]");
  }
};

ws.onerror = (e) => console.error("❌ Error:", e);
ws.onclose = () => console.log("🔌 Disconnected");
